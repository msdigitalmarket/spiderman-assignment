  <script>
    function showView(viewId) {
      document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active'));
      document.getElementById(viewId).classList.add('active');

      document.getElementById('nav-signup').classList.remove('active');
      document.getElementById('nav-admin').classList.remove('active');

      if (viewId === 'user-signup' || viewId === 'user-hero') {
        document.getElementById('nav-signup').classList.add('active');
      } else {
        document.getElementById('nav-admin').classList.add('active');
      }
    }

    function handleUserSignup(e) {
      e.preventDefault();
      const name = document.getElementById('user-name').value;
      const email = document.getElementById('user-email').value;
      
      const now = new Date();
      const formattedDate = now.toLocaleDateString() + ' ' + now.toLocaleTimeString();

      const newUser = { name, email, date: formattedDate };

      let users = JSON.parse(localStorage.getItem('apex_users') || '[]');
      users.push(newUser);

      localStorage.setItem('apex_users', JSON.stringify(users));

      document.getElementById('welcome-username').innerText = name;
      showView('user-hero');
    }

    function handleAdminLogin(e) {
      e.preventDefault();
      const email = document.getElementById('admin-email').value;
      const pass = document.getElementById('admin-pass').value;

      if (email === 'admin@apex.com' && pass === 'admin123') {
        loadAdminData();
        showView('admin-dashboard');
      } else {
        alert('Invalid Admin Credentials! Use admin@apex.com / admin123');
      }
    }

    function loadAdminData() {
      const users = JSON.parse(localStorage.getItem('apex_users') || '[]');
      const tbody = document.getElementById('user-table-body');
      tbody.innerHTML = '';

      if (users.length === 0) {
        tbody.innerHTML = '<tr><td colspan="4" style="text-align:center; color:#94A3B8;">No users registered yet.</td></tr>';
        return;
      }

      users.forEach((u, index) => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td>${index + 1}</td>
          <td>${u.name}</td>
          <td>${u.email}</td>
          <td>${u.date}</td>
        `;
        tbody.appendChild(tr);
      });
    }

    function logoutUser() {
      document.getElementById('user-name').value = '';
      document.getElementById('user-email').value = '';
      document.getElementById('user-pass').value = '';
      showView('user-signup');
    }

    function logoutAdmin() {
      document.getElementById('admin-email').value = '';
      document.getElementById('admin-pass').value = '';
      showView('admin-login');
    }
  </script>
